// Creating tables
Table Users {
  id            int     [pk, increment] // auto-increment
  name          string  [not null]
  email           string  [not null, unique]
  cpf            string  [not null, unique]       
  is_admin       boolean [default: false]
  active        boolean [default: true]
  created_at    timestamp
  updated_at    timestamp
  
}

// BLOCK MOTIVE!
Table Customers{
  id            int [pk, increment] // auto-increment
  created_at    timestamp
  updated_at    timestamp
  doc_ident      string    [not null, unique]      
  active        boolean   [default: true]
  block_motive  string    [default: null]
}

Table Orders {
  id              int     [pk, increment] // auto-increment
  salesman_id     integer [ref: > Users.id]
  customer_id       integer [ref: > Customers.id]
  product_id      integer [note: 'fk API']
  plan_id         integer [note: 'fk API']
  price           decimal [not null]
  discount        decimal [default: null]
  payment_mode    string  [default: null]
  status          integer [default: 0, note: '0=pending, 1=cancelled, 2=processed']
  cancel_motive   string  [default: null]
  created_at timestamp
  updated_at timestamp
}

Table Comissions {
  id            int [pk, increment] // auto-increment
  created_at    timestamp
  updated_at    timestamp
  salesman_id   integer   [ref: > Users.id]
  paid_at       timestamp [not null]
  amount        decimal   [not null]
  order_id      integer   [ref: - Orders.id]
  bonus_id      integer   [ref: - Bonus.id, default:null]
}

// THE BONUS IS ALWAYS THE BIGGEST ONE

Table Bonus {
  id                  int [ pk, increment] // auto-increment
  created_at          timestamp
  updated_at          timestamp
  description         string
  start_date          timestamp
  end_date            timestamp
  comission_perc      integer [note:'Percent of total sale']
  amount_limit        decimal [note: 'Total amount per salesman']
  active              boolean
  
}
